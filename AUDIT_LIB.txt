import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function updateSession(request: NextRequest) {
  const response = NextResponse.next({
  request,
  });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },

        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value, options }) => {
            request.cookies.set(name, value);
            response.cookies.set(name, value, options);
          });
        },
      },
    }
  );

  const {
    data: { user },
    error,
  } = await supabase.auth.getUser();

  const pathname = request.nextUrl.pathname;

  console.log("========== AEGRIS PROXY ==========");
  console.log("PATH:", pathname);
  console.log("COOKIE COUNT:", request.cookies.getAll().length);
  console.log("USER:", user?.email ?? "Ĺ˝ĂDNĂť UĹ˝IVATEL");
  console.log("ERROR:", error?.message ?? "null");
  console.log("===================================");

  const protectedRoutes = [
    "/dashboard",
    "/projects",
  ];

  const isProtectedRoute = protectedRoutes.some(
    (route) =>
      pathname === route ||
      pathname.startsWith(`${route}/`)
  );

  if (isProtectedRoute && !user) {
    const url = request.nextUrl.clone();

    url.pathname = "/login";
    url.searchParams.set("next", pathname);

    console.log(
      "AEGRIS PROXY: NEPĹIHLĂĹ ENĂť â†’ LOGIN"
    );

    return NextResponse.redirect(url);
  }

  return response;
}

function ndviHistoryTime(item: NdviHistory): number {
  const value = Date.parse(
    item.period_from || item.period_to || item.created_at
  );

  return Number.isFinite(value) ? value : NaN;
}

/**
 * JedinĂ˝ kanonickĂ˝ zdroj NDVI ÄŤasovĂ© Ĺ™ady pro engine i dashboard.
 * HistorickĂ© body po aktuĂˇlnĂ­ analĂ˝ze se ignorujĂ­ a aktuĂˇlnĂ­ analĂ˝za
 * se pĹ™idĂˇ jako poslednĂ­ bod pouze jednou.
 */
export function buildCanonicalNdviHistory(
  ndviHistory: NdviHistory[],
  currentNdvi: number | null,
  analysisCreatedAt: string | null = null
): NdviHistory[] {
  const analysisTime = analysisCreatedAt
    ? Date.parse(analysisCreatedAt)
    : NaN;

  const valid = ndviHistory
    .filter(
      (item) =>
        Number.isFinite(Number(item.ndvi)) &&
        Number.isFinite(ndviHistoryTime(item))
    )
    .filter(
      (item) =>
        !Number.isFinite(analysisTime) ||
        ndviHistoryTime(item) <= analysisTime
    )
    .sort(
      (a, b) =>
        ndviHistoryTime(a) - ndviHistoryTime(b)
    );

  const deduped: NdviHistory[] = [];

  for (const item of valid) {
    const previous = deduped[deduped.length - 1];

    if (
      previous &&
      Math.abs(
        ndviHistoryTime(previous) -
          ndviHistoryTime(item)
      ) < 1000
    ) {
      deduped[deduped.length - 1] = item;
    } else {
      deduped.push(item);
    }
  }

  if (
    currentNdvi == null ||
    !Number.isFinite(currentNdvi)
  ) {
    return deduped;
  }

  const currentTime = Number.isFinite(analysisTime)
    ? analysisTime
    : Date.now();

  const currentIso = new Date(currentTime).toISOString();
  const last = deduped[deduped.length - 1];

  if (
    last &&
    Math.abs(
      ndviHistoryTime(last) - currentTime
    ) < 1000 &&
    Math.abs(Number(last.ndvi) - currentNdvi) < 0.0005
  ) {
    return deduped;
  }

  return [
    ...deduped,
    {
      id: -1,
      project_id: deduped[0]?.project_id ?? 0,
      period_from: currentIso,
      period_to: currentIso,
      ndvi: currentNdvi,
      created_at: currentIso,
    },
  ];
}

export function calculateNdviTrend(
  ndviHistory: NdviHistory[],
  currentNdvi: number | null,
  analysisCreatedAt: string | null = null
): NdviTrend {
  if (
    currentNdvi == null ||
    !Number.isFinite(currentNdvi)
  ) {
    return {
      direction: "Nedostatek dat",
      slope: null,
      overallDelta: null,
      overallRelativeChangePct: null,
      latestChange: null,
      previousNdvi: null,
      points: 0,
    };
  }

  const canonical = buildCanonicalNdviHistory(
    ndviHistory,
    currentNdvi,
    analysisCreatedAt
  );

  const historical = canonical.filter(
    (item) => !(item.id === -1)
  );

  const previousNdvi = historical.length
    ? Number(historical[historical.length - 1].ndvi)
    : null;

  const values = historical
    .slice(-5)
    .map((item) => Number(item.ndvi))
    .concat(currentNdvi);

  if (values.length < 2) {
    return {
      direction: "Nedostatek dat",
      slope: null,
      overallDelta: null,
      overallRelativeChangePct: null,
      latestChange:
        previousNdvi == null
          ? null
          : currentNdvi - previousNdvi,
      previousNdvi,
      points: values.length,
    };
  }

  const n = values.length;
  const xMean = (n - 1) / 2;

  const yMean =
    values.reduce(
      (sum, value) => sum + value,
      0
    ) / n;

  let numerator = 0;
  let denominator = 0;

  values.forEach((value, index) => {
    const xDelta = index - xMean;

    numerator +=
      xDelta * (value - yMean);

    denominator +=
      xDelta * xDelta;
  });

  const slope =
    denominator > 0
      ? numerator / denominator
      : 0;

  const first = values[0];

  const overallDelta =
    currentNdvi - first;

  const overallRelativeChangePct =
    Math.abs(first) > 0
      ? (overallDelta / Math.abs(first)) * 100
      : 0;

  const latestChange =
    previousNdvi == null
      ? null
      : currentNdvi - previousNdvi;

  const direction =
    slope <= -0.015 ||
    overallRelativeChangePct <= -10
      ? "KlesajĂ­cĂ­"
      : slope >= 0.015 ||
          overallRelativeChangePct >= 10
        ? "RostoucĂ­"
        : "StabilnĂ­";

  return {
    direction,
    slope,
    overallDelta,
    overallRelativeChangePct,
    latestChange,
    previousNdvi,
    points: values.length,
  };
}

            : "OK";

      if (
        waterStatus ===
        "KritickĂ©"
      ) {
        criticalCount++;

        let detail =
          `Pro fĂˇzi ${growthStage || "neuvedenou"} je Kc ${cropCoefficient.toFixed(2)}. ` +

          `oÄŤekĂˇvanĂ© srĂˇĹľky jsou ${precip.toFixed(1)} mm. ` +

          `KontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je ${continuousWaterBalanceScore.toFixed(1)} bodu.`;

          weather?.soil_moisture_pct;

        const soilMin =
          cropProfile?.soil_moisture_min_pct;

        const soilMax =
          cropProfile?.soil_moisture_max_pct;

        if (
          soil != null &&
          soilMin != null &&
          soilMax != null
        ) {
          detail +=
            ` PĹŻdnĂ­ vlhkost ${soil.toFixed(1)} % je ` +
            `${
              soil < soilMin
                ? "pod"
                : soil > soilMax
                  ? "nad"
                  : "v"
            } ` +
            `doporuÄŤenĂ˝m rozsahem ${soilMin.toFixed(1)}â€“${soilMax.toFixed(1)} %.`;
        }

        if (
          soil != null &&
          soilProfile?.wilting_point_pct != null &&
          soil <=
            soilProfile.wilting_point_pct
        ) {
          detail +=
            ` SouÄŤasnÄ› je pĹŻdnĂ­ vlhkost ${soil.toFixed(1)} % ` +
            `na/pod bodem vadnutĂ­ ${soilProfile.wilting_point_pct.toFixed(1)} %.`;
        }

        factors.push({
          label: "VodnĂ­ bilance",
          status: "KritickĂ©",

        });

        actions.push(
          "ProvÄ›Ĺ™it vodnĂ­ reĹľim porostu a skuteÄŤnou pĹŻdnĂ­ vlhkost; kontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je v kritickĂ©m pĂˇsmu."
        );
      } else if (
        waterStatus ===
        "UpozornÄ›nĂ­"
      ) {
        warningCount++;

        let detail =
          `Pro fĂˇzi ${growthStage || "neuvedenou"} je Kc ${cropCoefficient.toFixed(2)}. ` +

          weather?.soil_moisture_pct;

        const soilMin =
          cropProfile?.soil_moisture_min_pct;

          cropProfile?.soil_moisture_max_pct;

          soilMin != null &&
          soilMax != null
        ) {

            detail +=
              ` PĹŻdnĂ­ vlhkost ${soil.toFixed(1)} % je zatĂ­m v doporuÄŤenĂ©m rozsahu ${soilMin.toFixed(1)}â€“${soilMax.toFixed(1)} %.`;
          } else {
            detail +=
              ` PĹŻdnĂ­ vlhkost ${soil.toFixed(1)} % je nad doporuÄŤenĂ˝m maximem ${soilMax.toFixed(1)} %.`;

            " AktuĂˇlnĂ­ pĹŻdnĂ­ vlhkost nelze porovnat s doporuÄŤenĂ˝m rozsahem.";

          label: "VodnĂ­ bilance",
          status: "UpozornÄ›nĂ­",
          detail,
        });

        actions.push(
          "Sledovat vodnĂ­ reĹľim porostu; kontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je ve varovnĂ©m pĂˇsmu."

      } else {

          label: "VodnĂ­ bilance",
          status: "OK",
          detail:
            `RĹŻstovĂˇ fĂˇze ${growthStage || "neuvedenĂˇ"} mĂˇ Kc ${cropCoefficient.toFixed(2)}. ` +
            `ETâ‚€ ${et0.toFixed(1)} mm/den, orientaÄŤnĂ­ ETc ${cropEt.toFixed(1)} mm/den, ` +
            `oÄŤekĂˇvanĂ© srĂˇĹľky ${precip.toFixed(1)} mm a kontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance ` +

        });
      }
    } else if (
      Number.isFinite(precip)
    ) {
      // Fallback pĹ™i absenci ETâ‚€.
      // I zde pouĹľĂ­vĂˇme stejnĂ© hranice stavu jako u plnĂ©ho vĂ˝poÄŤtu,

      continuousWaterBalanceScore =
        clampScore(
          precip >= 5
            ? 100
            : precip >= 1
              ? 70 +
                ((precip - 1) /
                  4) *
                  30
              : precip * 70
        );

      const waterStatus =
        continuousWaterBalanceScore < 30
          ? "KritickĂ©"
          : continuousWaterBalanceScore < 60
            ? "UpozornÄ›nĂ­"
            : "OK";

      if (
        waterStatus ===
        "KritickĂ©"
      ) {
        criticalCount++;

        factors.push({
          label: "VodnĂ­ bilance",
          status: "KritickĂ©",
          detail:
            `Za 24 h se oÄŤekĂˇvĂˇ pouze ${precip.toFixed(1)} mm srĂˇĹľek. ` +
            `Pro pĹ™esnÄ›jĹˇĂ­ vĂ˝poÄŤet chybĂ­ ETâ‚€; Kc rĹŻstovĂ© fĂˇze je ${cropCoefficient.toFixed(2)}. ` +
            `KontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je ${continuousWaterBalanceScore.toFixed(1)} bodu.`,
        });

        actions.push(
          "ProvÄ›Ĺ™it vodnĂ­ reĹľim porostu; oÄŤekĂˇvanĂ© srĂˇĹľky jsou nĂ­zkĂ© a bez ETâ‚€ nelze pĹ™esnÄ› urÄŤit vodnĂ­ deficit."
        );
      } else if (
        waterStatus ===
        "UpozornÄ›nĂ­"
      ) {
        warningCount++;

        factors.push({
          label: "VodnĂ­ bilance",
          status: "UpozornÄ›nĂ­",
          detail:
            `Za 24 h se oÄŤekĂˇvĂˇ ${precip.toFixed(1)} mm srĂˇĹľek. ` +
            `Pro pĹ™esnÄ›jĹˇĂ­ vĂ˝poÄŤet chybĂ­ ETâ‚€; Kc rĹŻstovĂ© fĂˇze je ${cropCoefficient.toFixed(2)}. ` +
            `KontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je ${continuousWaterBalanceScore.toFixed(1)} bodu.`,
        });

        actions.push(
          "Sledovat vodnĂ­ reĹľim porostu; pro pĹ™esnÄ›jĹˇĂ­ vĂ˝poÄŤet chybĂ­ ETâ‚€."
        );
      } else {
        factors.push({
          label: "VodnĂ­ bilance",
          status: "OK",
          detail:
            `OÄŤekĂˇvanĂ© srĂˇĹľky za 24 h: ${precip.toFixed(1)} mm. ` +
            `Pro pĹ™esnÄ›jĹˇĂ­ vodnĂ­ bilanci chybĂ­ ETâ‚€; Kc rĹŻstovĂ© fĂˇze je ${cropCoefficient.toFixed(2)}. ` +
            `KontinuĂˇlnĂ­ skĂłre vodnĂ­ bilance je ${continuousWaterBalanceScore.toFixed(1)} bodu.`,
        });
      }
    }
  } else {
    factors.push({
      label: "VodnĂ­ bilance",
      status: "N/A",
      detail:
        "PĹ™edpovÄ›ÄŹ srĂˇĹľek nenĂ­ k dispozici.",
    });
  }

  // ---------------------------------------------------------
  // 6. PravdÄ›podobnost srĂˇĹľek, pokud ji API poskytne
  // ---------------------------------------------------------
  if (
    weather?.precipitation_probability_pct !=
    null
  ) {
    const probability =
      weather.precipitation_probability_pct;

    factors.push({
      label: "PravdÄ›podobnost srĂˇĹľek",
      status: "OK",
      detail:
        `PravdÄ›podobnost srĂˇĹľek je ${probability.toFixed(0)} %.`,
    });
  }

  // ---------------------------------------------------------
  // KombinovanĂ© riziko: souÄŤasnÄ› nĂ­zkĂ© NDVI + vodnĂ­/tepelnĂ˝ stres
  // ---------------------------------------------------------
  const hasLowNdvi =
    ndvi < target * 0.8;

  const hasWaterWarning =
    factors.some(
      (factor) =>
        factor.label ===
          "VodnĂ­ bilance" &&
        (
          factor.status ===
            "UpozornÄ›nĂ­" ||
          factor.status ===
            "KritickĂ©"
        )
    );

  const hasCurrentTemperatureWarning =
    factors.some(
      (factor) =>
        factor.label ===
          "Teplota" &&
        (
          factor.status ===
            "UpozornÄ›nĂ­" ||
          factor.status ===
            "KritickĂ©"
        )
    );

  const hasForecastTemperatureRisk =
    factors.some(
      (factor) =>
        factor.label ===
          "TeplotnĂ­ vĂ˝hled 24 h" &&
        (
          factor.status ===
            "UpozornÄ›nĂ­" ||
          factor.status ===
            "KritickĂ©"
        )
    );

  if (
    hasLowNdvi &&
    hasWaterWarning &&
    hasCurrentTemperatureWarning
  ) {
    criticalCount++;

    factors.push({
      label: "KombinovanĂ© riziko",
      status: "KritickĂ©",
      detail:
        "SouÄŤasnÄ› je zaznamenĂˇna nĂ­zkĂˇ vegetaÄŤnĂ­ aktivita a nepĹ™Ă­znivĂˇ vodnĂ­ i teplotnĂ­ situace. Jde o kombinovanĂ˝ signĂˇl, kterĂ˝ vyĹľaduje terĂ©nnĂ­ ovÄ›Ĺ™enĂ­.",
    });

    actions.push(
      "ProvĂ©st prioritnĂ­ terĂ©nnĂ­ kontrolu porostu a ovÄ›Ĺ™it vodnĂ­ a teplotnĂ­ stres pĹ™ed rozhodnutĂ­m o zĂˇsahu."
    );
  }

  // ---------------------------------------------------------
  // AEGRIS SCORE 0â€“100 â€” KANONICKĂť MODEL
  // ---------------------------------------------------------
  //
  // VĂˇhy:
  // NDVI 25 % | Trend NDVI 20 % | VodnĂ­ bilance 20 %
  // Teplota 10 % | TeplotnĂ­ vĂ˝hled 10 % | Vlhkost pĹŻdy 15 %
  // CELKEM 100 %.
  //
  // Status faktoru a kontinuĂˇlnĂ­ score jsou oddÄ›lenĂ©.
  // KombinovanĂ© riziko se do score nepĹ™iÄŤĂ­tĂˇ podruhĂ©.
  //
  // VodnĂ­ bilance a pĹŻdnĂ­ vlhkost jsou jiĹľ vypoÄŤtenĂ© vĂ˝Ĺˇe:
  // zde se pouze pouĹľijĂ­ jejich kanonickĂ© hodnoty.
  // KaĹľdĂ˝ signĂˇl se do vĂ˝slednĂ©ho score zapoÄŤĂ­tĂˇvĂˇ prĂˇvÄ› jednou.

  const scoreInterpolate = (
    value: number,
    points: Array<[number, number]>
  ): number => {
    if (!Number.isFinite(value)) {
      return 0;
    }

    if (value <= points[0][0]) {
      return points[0][1];
    }

    for (let i = 1; i < points.length; i++) {
      const [x1, y1] = points[i - 1];
      const [x2, y2] = points[i];

      if (value <= x2) {
        const ratio =
          (value - x1) /
          Math.max(x2 - x1, 0.000001);

        return y1 + (y2 - y1) * ratio;
      }
    }

    return points[points.length - 1][1];
  };

  // 1. NDVI â€” plynulĂ© skĂłre podle pomÄ›ru k cĂ­li fĂˇze.
  const ndviRatio =
    target > 0
      ? ndvi / target
      : 0;

  const continuousNdviScore =
    clampScore(
      scoreInterpolate(
        ndviRatio,
        [
          [0.40, 0],
          [0.50, 20],
          [0.65, 45],
          [0.80, 70],
          [0.90, 85],
          [1.00, 100],
        ]
      )
    );

  // 2. Trend NDVI â€” 60 % relativnĂ­ zmÄ›na, 40 % slope.
  let continuousTrendScore:
    number | null = null;

  if (trendHistory.length >= 2) {
    const values = [
      ...trendHistory.map(
        (item) => Number(item.ndvi)
      ),
      ndvi,
    ].filter(Number.isFinite);

    const n = values.length;
    const xMean = (n - 1) / 2;

    const yMean =
      values.reduce(
        (sum, value) => sum + value,
        0
      ) / n;

    let numerator = 0;
    let denominator = 0;

    values.forEach(
      (value, index) => {
        const xDelta =
          index - xMean;

        numerator +=
          xDelta * (value - yMean);

        denominator +=
          xDelta * xDelta;
      }
    );

    const slope =
      denominator > 0
        ? numerator / denominator
        : 0;

    const first = values[0];

    const relativeChangePct =
      Math.abs(first) > 0
        ? ((ndvi - first) /
            Math.abs(first)) *
          100
        : 0;

    const relativeScore =
      scoreInterpolate(
        relativeChangePct,
        [
          [-30, 0],
          [-20, 30],
          [-10, 60],
          [0, 80],
          [10, 90],
          [20, 100],
        ]
      );

    const slopeScore =
      scoreInterpolate(
        slope,
        [
          [-0.05, 0],
          [-0.03, 30],
          [-0.015, 60],
          [0, 80],
          [0.015, 90],
          [0.03, 100],
        ]
      );

    continuousTrendScore =
      clampScore(
        relativeScore * 0.6 +
        slopeScore * 0.4
      );
  } else if (
    previousNdvi != null &&
    Number.isFinite(previousNdvi)
  ) {
    const relativeChangePct =
      Math.abs(previousNdvi) > 0
        ? ((ndvi - previousNdvi) /
            Math.abs(previousNdvi)) *
          100
        : 0;

    continuousTrendScore =
      clampScore(
        scoreInterpolate(
          relativeChangePct,
          [
            [-30, 0],
            [-20, 30],
            [-10, 60],
            [0, 80],
            [10, 90],
            [20, 100],
          ]
        )
      );
  }

  // 3. VodnĂ­ bilance â€” hodnota je vypoÄŤtena jednou vĂ˝Ĺˇe.
  // Pokud nejsou ET0 data, pouĹľije se jejĂ­ existujĂ­cĂ­ fallback.
  const waterBalanceScore =
    continuousWaterBalanceScore;

  // 4. AktuĂˇlnĂ­ teplota.
  let continuousTemperatureScore:
    number | null = null;

  if (
    weather?.temperature_c != null &&
    stageMinTemperature != null &&
    stageMaxTemperature != null &&
    Number.isFinite(
      Number(weather.temperature_c)
    )
  ) {
    const temperature =
      Number(weather.temperature_c);

    const deviation =
      temperature < stageMinTemperature
        ? stageMinTemperature - temperature
        : temperature > stageMaxTemperature
          ? temperature - stageMaxTemperature
          : 0;

    continuousTemperatureScore =
      clampScore(
        scoreInterpolate(
          deviation,
          [
            [0, 100],
            [1, 85],
            [2, 70],
            [3, 55],
            [4, 35],
            [6, 0],
          ]
        )
      );
  }

  // 5. TeplotnĂ­ vĂ˝hled â€” mĂ­rnÄ›jĹˇĂ­ neĹľ skuteÄŤnÄ› namÄ›Ĺ™enĂˇ teplota.
  let continuousForecastTemperatureScore:
    number | null = null;

  if (
    weather?.next24h_min_temperature_c != null &&
    weather?.next24h_max_temperature_c != null &&
    stageMinTemperature != null &&
    stageMaxTemperature != null &&
    Number.isFinite(
      Number(
        weather.next24h_min_temperature_c
      )
    ) &&
    Number.isFinite(
      Number(
        weather.next24h_max_temperature_c
      )

  ) {
    const forecastMin =
      Number(
        weather.next24h_min_temperature_c
      );

    const forecastMax =
      Number(
        weather.next24h_max_temperature_c
      );

    const deviation =
      Math.max(
        stageMinTemperature -
          forecastMin,
        forecastMax -
          stageMaxTemperature,
        0
      );

    continuousForecastTemperatureScore =
      clampScore(
        scoreInterpolate(
          deviation,
          [
            [0, 100],
            [1, 95],
            [2, 85],
            [3, 70],
            [4, 55],
            [6, 20],
          ]
        )
      );
  }

  // 6. Vlhkost pĹŻdy.
  // Preferujeme fyzikĂˇlnĂ­ profil projektu (wilting point -> 0,
  // field capacity -> 100). Pokud nenĂ­ k dispozici, pouĹľijeme
  // doporuÄŤenĂ˝ rozsah plodiny. U fallbacku je celĂ˝ doporuÄŤenĂ˝
  // rozsah hodnocen jako pĹ™Ă­znivĂ˝; mimo nÄ›j skĂłre klesĂˇ.
  let continuousSoilMoistureScore:
    number | null = null;

  if (
    weather?.soil_moisture_pct != null &&
    soilProfile?.wilting_point_pct != null &&
    soilProfile?.field_capacity_pct != null &&
    Number.isFinite(
      Number(weather.soil_moisture_pct)
    ) &&
    Number.isFinite(
      Number(soilProfile.wilting_point_pct)
    ) &&
    Number.isFinite(
      Number(soilProfile.field_capacity_pct)
    ) &&
    Number(
      soilProfile.field_capacity_pct
    ) >
      Number(
        soilProfile.wilting_point_pct
      )
  ) {
    const soil =
      Number(weather.soil_moisture_pct);

    const wp =
      Number(soilProfile.wilting_point_pct);

    const fc =
      Number(soilProfile.field_capacity_pct);

    continuousSoilMoistureScore =
      clampScore(
        ((soil - wp) /
          (fc - wp)) *
          100
      );
  } else if (
    weather?.soil_moisture_pct != null &&
    cropProfile?.soil_moisture_min_pct != null &&
    cropProfile?.soil_moisture_max_pct != null &&
    Number.isFinite(
      Number(weather.soil_moisture_pct)
    ) &&
    Number.isFinite(
      Number(cropProfile.soil_moisture_min_pct)
    ) &&
    Number.isFinite(
      Number(cropProfile.soil_moisture_max_pct)
    ) &&
    Number(
      cropProfile.soil_moisture_max_pct
    ) >
      Number(
        cropProfile.soil_moisture_min_pct
      )
  ) {
    const soil =
      Number(weather.soil_moisture_pct);

    const min =
      Number(cropProfile.soil_moisture_min_pct);

    const max =
      Number(cropProfile.soil_moisture_max_pct);

    const range =
      max - min;

    if (soil >= min && soil <= max) {
      continuousSoilMoistureScore = 100;
    } else {
      const distance =
        soil < min
          ? min - soil
          : soil - max;

      // Jeden celĂ˝ rozsah mimo doporuÄŤenĂ© pĂˇsmo odpovĂ­dĂˇ
      // poklesu o 70 bodĹŻ; hodnota zĹŻstĂˇvĂˇ plynulĂˇ.
      continuousSoilMoistureScore =
        clampScore(
          100 -
            (distance / range) * 70
        );
    }
  }

  // KanonickĂ© faktory â€” pouze tÄ›chto Ĺˇest mĂˇ vĂˇhu v AEGRIS SCORE.
  const weightedFactors = [
    {
      label: "NDVI",
      weight: 25,
      value: continuousNdviScore,
    },
    {
      label: "Trend NDVI",
      weight: 20,
      value: continuousTrendScore,
    },
    {
      label: "VodnĂ­ bilance",
      weight: 20,
      value: waterBalanceScore,
    },
    {
      label: "Teplota",
      weight: 10,
      value: continuousTemperatureScore,
    },
    {
      label: "TeplotnĂ­ vĂ˝hled 24 h",
      weight: 10,
      value: continuousForecastTemperatureScore,
    },
    {
      label: "Vlhkost pĹŻdy",
      weight: 15,
      value: continuousSoilMoistureScore,
    },
  ];

  const availableFactors =
    weightedFactors.filter(
      (factor) =>
        factor.value != null &&
        Number.isFinite(
          Number(factor.value)
        )
    );

  const totalWeight =
    availableFactors.reduce(
      (sum, factor) =>
        sum + factor.weight,
      0
    );

  const weightedScore =
    totalWeight > 0
      ? availableFactors.reduce(
          (sum, factor) =>
            sum +
            Number(factor.value) *
              factor.weight,
          0
        ) / totalWeight
      : 0;

  const score =
    Math.round(
      clampScore(weightedScore)
    );

  const scoreBreakdown:
    ScoreBreakdownItem[] =
    availableFactors.map(
      (factor) => ({
        label: factor.label,
        score: Math.round(
          Number(factor.value)
        ),
        weight: factor.weight,
        contribution:
          Math.round(
            (
              (Number(factor.value) *
                factor.weight) /
              totalWeight
            ) *
              100
          ) / 100,
      })
    );

  const dataCompletenessPct =
    Math.round(
      (availableFactors.length /
        weightedFactors.length) *
        100
    );

  const scoreLevel:
    ContextEvaluation["scoreLevel"] =
    score >= 85
      ? "Velmi dobrĂ˝ stav"
      : score >= 70
        ? "DobrĂ˝ stav"
        : score >= 55
          ? "ZvĂ˝ĹˇenĂ© riziko"
          : score >= 40
            ? "VysokĂ© riziko"
            : "KritickĂ˝ stav";

  let level:
    ContextEvaluation["level"];

  if (score >= 70) {
    level =
      warningCount > 0
        ? "UpozornÄ›nĂ­"
        : "OptimĂˇlnĂ­";
  } else if (score >= 40) {
    level = "UpozornÄ›nĂ­";
  } else {
    level = "KritickĂ©";
  }

  const cropLabel =
    cropProfile?.name ??
    "plodina";

  const stageLabel =
    growthStage ||
    "aktuĂˇlnĂ­ rĹŻstovĂˇ fĂˇze";

  let summary =
    level === "KritickĂ©"
      ? `AEGRIS identifikoval kritickĂ˝ stav projektu s plodinou ${cropLabel}.`
      : level === "UpozornÄ›nĂ­"
        ? `AEGRIS identifikoval podmĂ­nku, kterou je vhodnĂ© u ${cropLabel} sledovat.`
        : `AktuĂˇlnÄ› dostupnĂˇ data jsou pro ${cropLabel} bez vĂ˝znamnĂ©ho varovnĂ©ho signĂˇlu.`;

  if (hasLowNdvi) {
    summary +=
      ` NDVI je pod orientaÄŤnĂ­ ĂşrovnĂ­ pro fĂˇzi ${stageLabel}.`;
  } else if (
    previousNdvi != null &&
    ndvi > previousNdvi
  ) {
    summary +=
      " NDVI je oproti pĹ™edchozĂ­mu mÄ›Ĺ™enĂ­ vyĹˇĹˇĂ­.";
  } else if (
    previousNdvi != null &&
    ndvi < previousNdvi
  ) {
    summary +=
      " NDVI je oproti pĹ™edchozĂ­mu mÄ›Ĺ™enĂ­ niĹľĹˇĂ­.";
  }

  let recommendation =
    level === "KritickĂ©"
      ? "Projekt vyĹľaduje zvĂ˝Ĺˇenou pozornost. Nejprve ovÄ›Ĺ™te kritickĂ© faktory pĹ™Ă­mo v terĂ©nu a nĂˇslednÄ› rozhodnÄ›te o dalĹˇĂ­m zĂˇsahu."
      : level === "UpozornÄ›nĂ­"
        ? "Projekt je monitorovatelnĂ˝, ale nÄ›kterĂ˝ z dostupnĂ˝ch ukazatelĹŻ vyĹľaduje pozornost. Sledujte jeho vĂ˝voj a ovÄ›Ĺ™te skuteÄŤnĂ˝ stav porostu."
        : "Projekt je podle aktuĂˇlnÄ› dostupnĂ˝ch dat v pĹ™Ă­znivĂ©m stavu. PokraÄŤujte v pravidelnĂ©m monitoringu a sledujte zmÄ›nu trendu.";

  if (hasWaterWarning) {
    recommendation +=
      " VodnĂ­ reĹľim vyĹľaduje zvĂ˝Ĺˇenou pozornost vzhledem ke kombinaci dostupnĂ© vlhkosti, oÄŤekĂˇvanĂ˝ch srĂˇĹľek a evapotranspirace.";
  }

  if (hasCurrentTemperatureWarning) {
    recommendation +=
      " AktuĂˇlnĂ­ teplotnĂ­ podmĂ­nky jsou mimo doporuÄŤenĂ˝ rozsah plodiny a vyĹľadujĂ­ ovÄ›Ĺ™enĂ­ v terĂ©nu.";
  }

  if (hasForecastTemperatureRisk) {
    recommendation +=
      " TeplotnĂ­ vĂ˝hled na dalĹˇĂ­ch 24 hodin pĹ™edstavuje pĹ™edpoklĂˇdanĂ© riziko, kterĂ© je vhodnĂ© sledovat.";
  }

  if (actions.length === 0) {
    actions.push(
      "PokraÄŤovat v pravidelnĂ©m monitoringu a porovnat dalĹˇĂ­ NDVI analĂ˝zu s aktuĂˇlnĂ­m stavem projektu."
    );
  }

  const priority:
    ContextEvaluation["priority"] =
    level === "KritickĂ©"
      ? "KritickĂˇ"
      : criticalCount > 0
        ? "VysokĂˇ"
        : warningCount >= 2
          ? "VysokĂˇ"
          : warningCount === 1
            ? "StĹ™ednĂ­"
            : "NĂ­zkĂˇ";

  return {
    level,
    priority,
    score,
    scoreLevel,
    trend: ndviTrend,
    scoreBreakdown,
    criticalFactorCount:
      criticalCount,
    warningFactorCount:
      warningCount,
    evaluatedFactorCount:
      evaluatedCount,
    dataCompletenessPct,
    summary,
    recommendation,
    actions: Array.from(
      new Set(actions)
    ).slice(0, 5),
    factors,
  };
}
